cmake_minimum_required(VERSION 3.25)
# Embed MSVC debug symbols in objects without a persistent compiler PDB server.
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
project(HarranCraft LANGUAGES CXX)
include(CTest)

add_library(harrancraft_wire INTERFACE)
target_include_directories(harrancraft_wire INTERFACE include)
target_compile_features(harrancraft_wire INTERFACE cxx_std_20)

function(harrancraft_checks target)
    target_link_libraries(${target} PRIVATE harrancraft_wire)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wsign-conversion)
    endif()
endfunction()

if(WIN32)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The host probe requires Windows x64")
    endif()
    add_library(hc-host-probe SHARED native/host/host_probe_windows.cpp)
    target_compile_definitions(hc-host-probe PRIVATE HC_HOST_PROBE_BUILD=1)
    set_target_properties(hc-host-probe PROPERTIES PREFIX "" OUTPUT_NAME "hc-host-probe")
    harrancraft_checks(hc-host-probe)
    target_link_libraries(hc-host-probe PRIVATE bcrypt)
endif()

if(BUILD_TESTING)
    add_executable(hc-wire-test native/tests/wire_test.cpp)
    harrancraft_checks(hc-wire-test)
    add_test(NAME wire-contract COMMAND hc-wire-test)
    add_executable(hc-host-observation-test native/tests/host_observation_test.cpp)
    harrancraft_checks(hc-host-observation-test)
    add_test(NAME host-observation COMMAND hc-host-observation-test)
    add_executable(hc-host-build-test native/tests/host_build_test.cpp)
    harrancraft_checks(hc-host-build-test)
    add_test(NAME host-build COMMAND hc-host-build-test)
    add_executable(hc-bridge-peer native/tests/bridge_peer.cpp)
    harrancraft_checks(hc-bridge-peer)
    add_executable(hc-pipe-peer native/tests/pipe_peer.cpp)
    harrancraft_checks(hc-pipe-peer)
    if(WIN32)
        add_executable(hc-host-probe-test native/tests/host_probe_windows_test.cpp)
        harrancraft_checks(hc-host-probe-test)
        add_dependencies(hc-host-probe-test hc-host-probe)
        add_test(NAME host-probe-process COMMAND hc-host-probe-test)
    endif()
endif()
